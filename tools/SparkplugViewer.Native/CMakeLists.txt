cmake_minimum_required(VERSION 3.20)
# The bridge owns all C++ allocations; no additional VC runtime installation
# is required by this DLL in the framework-dependent Viewer package.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(SparkplugViewerNative LANGUAGES CXX)
set(SPV_SHOWINCLUDES_PREFIX "" CACHE STRING "Correctly decoded localized MSVC dependency prefix")
if(SPV_SHOWINCLUDES_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SPV_SHOWINCLUDES_PREFIX}")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Build engine research suites" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Sparkplug" engine)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Winx" game)
add_library(SparkplugViewerNative SHARED ViewerBridge.cpp ResourceGraph.cpp RenderMeshView.cpp NavigationView.cpp SpatialView.cpp LightInspection.cpp)
target_link_libraries(SparkplugViewerNative PRIVATE WinxGameCore)
target_compile_features(SparkplugViewerNative PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(SparkplugViewerNative PRIVATE /W4 /permissive- /utf-8)
endif()
option(SPV_BUILD_CHECKS "Build selected existing engine checks for the viewing core" OFF)
if(SPV_BUILD_CHECKS)
  enable_testing()
  foreach(suite IN ITEMS AnimationKey AnimationRuntime NodeWorld TransformInput SanReader CollisionCore MeshBVCore FullLoader TextureSerialization DataBlockWriter ReadReference NodeSerialization MeshReader RenderNode StaticRenderObject MaterialSerialization MaterialController MaterialColor SkinSerialization SkinRender UVFunction ColorFunction SpatialSerialization SceneSerialization SkyBox NavigationSerialization LensFlare OcclusionTopology ParticleSerialization LightSerialization)
    add_executable(Viewer${suite}Checks "../../Sparkplug/Tests/sp${suite}Tests.cpp")
    target_link_libraries(Viewer${suite}Checks PRIVATE WinxGameCore)
    add_test(NAME Viewer${suite}Checks COMMAND Viewer${suite}Checks)
    set_tests_properties(Viewer${suite}Checks PROPERTIES TIMEOUT 20)
  endforeach()
endif()
